// ===================================================== // FXEC APPS SCRIPT BRIDGE // ===================================================== const pendingRequests = {}; let requestCounter = 0; let bridgeLoaded = false; let bridgeQueue = []; const bridge = document.getElementById('backendBridge'); // ----------------------------------------------------- // BRIDGE IFRAME LOADED // ----------------------------------------------------- bridge.addEventListener('load', function () { console.log('FXEC Apps Script Bridge loaded'); bridgeLoaded = true; // Update connection message const status = document.getElementById('serverStatus'); if (status) { status.textContent = '✓ FXEC server connected'; status.className = 'server-status connected'; } // Send all requests waiting for the bridge const queue = bridgeQueue; bridgeQueue = []; queue.forEach(function (request) { sendToBridge(request); }); // Also send a PING bridge.contentWindow.postMessage( { source: 'FXEC_MOCK_INTERVIEW', action: 'PING' }, '*' ); }); // ----------------------------------------------------- // SEND REQUEST TO BRIDGE // ----------------------------------------------------- function sendToBridge(request) { if (!bridge || !bridge.contentWindow) { console.error('Bridge iframe is not available'); return; } console.log('Sending to FXEC Bridge:', request.action); bridge.contentWindow.postMessage( request, '*' ); } // ----------------------------------------------------- // BACKEND REQUEST // ----------------------------------------------------- function backendRequest(action, payload) { return new Promise(function (resolve, reject) { const requestId = 'REQ-' + Date.now() + '-' + (++requestCounter); pendingRequests[requestId] = { resolve: resolve, reject: reject }; const request = { source: 'FXEC_MOCK_INTERVIEW', requestId: requestId, action: action, payload: payload || {} }; // If iframe is already loaded, // send immediately. if (bridgeLoaded) { sendToBridge(request); } else { // Otherwise wait until iframe loads. console.log( 'Bridge not loaded yet. Queueing:', action ); bridgeQueue.push(request); } // 30-second timeout setTimeout(function () { if (pendingRequests[requestId]) { delete pendingRequests[requestId]; reject( new Error( 'FXEC server did not respond. Please refresh the page and try again.' ) ); } }, 30000); }); } // ----------------------------------------------------- // RECEIVE RESPONSE FROM APPS SCRIPT // ----------------------------------------------------- window.addEventListener('message', function (event) { const data = event.data; if (!data) { return; } if ( data.source !== 'FXEC_MOCK_INTERVIEW_BRIDGE' ) { return; } // ------------------------------------------------- // BRIDGE READY // ------------------------------------------------- if (data.type === 'BRIDGE_READY') { console.log( 'FXEC Bridge says READY' ); return; } // ------------------------------------------------- // RESPONSE // ------------------------------------------------- if (data.type !== 'RESPONSE') { return; } const request = pendingRequests[data.requestId]; if (!request) { console.warn( 'Unknown request:', data.requestId ); return; } delete pendingRequests[data.requestId]; if (data.success) { console.log( 'FXEC backend response received:', data ); request.resolve(data.data); } else { console.error( 'FXEC backend error:', data.error ); request.reject( new Error( data.error || 'FXEC backend request failed.' ) ); } });
